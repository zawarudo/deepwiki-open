### 1. Create Error Classes in `api/embedding_errors.py`
```python
from typing import List, Dict, Optional
from dataclasses import dataclass

@dataclass
class EmbeddingError:
    """Structured embedding error information"""
    document_id: str
    document_snippet: str
    error_type: str
    error_message: str
    suggested_action: str
    timestamp: str

class EmbeddingGenerationError(Exception):
    """Enhanced exception with structured error info"""
    def __init__(self, 
                 message: str,
                 errors: List[EmbeddingError] = None,
                 summary: Dict = None):
        super().__init__(message)
        self.errors = errors or []
        self.summary = summary or {}
    
    def get_summary(self) -> str:
        """Get human-readable error summary"""
        if self.summary:
            return (f"Embedding Generation Failed: "
                   f"{self.summary.get('failed', 0)}/{self.summary.get('total', 0)} failed. "
                   f"Reasons: {', '.join(self.summary.get('reasons', []))}")
        return str(self)
```

### 2. Enhance GoogleEmbeddingClient Error Handling
```python
def embed_batch(self, texts: List[str], doc_ids: List[str] = None) -> EmbedderOutput:
    """Enhanced batch embedding with detailed error reporting"""
    doc_ids = doc_ids or [f"doc_{i}" for i in range(len(texts))]
    errors = []
    embeddings = []
    
    for idx, (text, doc_id) in enumerate(zip(texts, doc_ids)):
        try:
            embedding = self._generate_single_embedding(text)
            embeddings.append(embedding)
        except Exception as e:
            error = self._create_error_info(e, text, doc_id, idx)
            errors.append(error)
            logger.error(f"Failed to embed {doc_id}: {error.error_message}")
    
    if errors and not embeddings:
        raise EmbeddingGenerationError(
            message="All embeddings failed",
            errors=errors,
            summary=self._create_error_summary(errors, len(texts))
        )
    
    return EmbedderOutput(
        embeddings=embeddings,
        errors=errors,
        summary=self._create_summary(len(embeddings), len(errors))
    )

def _create_error_info(self, exception: Exception, text: str, doc_id: str, idx: int) -> EmbeddingError:
    """Create structured error information"""
    error_type = type(exception).__name__
    snippet = text[:100] + "..." if len(text) > 100 else text
    
    suggested_actions = {
        "RateLimitError": "Wait 60 seconds before retrying",
        "QuotaExceededError": "Check API quota in Google Cloud Console",
        "TimeoutError": "Retry with smaller batch size",
        "InvalidRequestError": "Check document format and content",
    }
    
    return EmbeddingError(
        document_id=doc_id,
        document_snippet=snippet,
        error_type=error_type,
        error_message=str(exception),
        suggested_action=suggested_actions.get(error_type, "Check logs for details"),
        timestamp=datetime.now().isoformat()
    )
```

### 3. Enhanced Logging
```python
import structlog

logger = structlog.get_logger()

def log_batch_results(self, success_count: int, failure_count: int, errors: List[EmbeddingError]):
    """Log detailed batch processing results"""
    logger.info(
        "Batch embedding completed",
        success_count=success_count,
        failure_count=failure_count,
        total=success_count + failure_count,
        success_rate=f"{(success_count/(success_count+failure_count)*100):.1f}%"
    )
    
    if errors:
        error_summary = {}
        for error in errors:
            error_summary[error.error_type] = error_summary.get(error.error_type, 0) + 1
        
        logger.warning(
            "Embedding failures detected",
            error_types=error_summary,
            failed_documents=[e.document_id for e in errors[:10]],  # First 10
            suggested_actions=list(set(e.suggested_action for e in errors))
        )
```
