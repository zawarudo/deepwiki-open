import time
from typing import Optional, Callable
from functools import wraps

def exponential_backoff_retry(
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 8.0,
    retryable_exceptions: tuple = (Exception,)
):
    """Decorator for exponential backoff retry logic"""
    def decorator(func: Callable):
        @wraps(func)
        def wrapper(*args, **kwargs):
            delay = base_delay
            last_exception = None
            
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except retryable_exceptions as e:
                    last_exception = e

                        logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay}s...")
                        time.sleep(delay)

                    else:
                        logger.error(f"All {max_retries + 1} attempts failed")
                        raise
            
            raise last_exception
        return wrapper
    return decorator
```

def embed_batch_with_retry(self, texts: List[str]) -> List[List[float]]:
    """Embed batch with retry for failed items"""
    embeddings = [None] * len(texts)
    failed_indices = list(range(len(texts)))
    
    for attempt in range(self.max_retries + 1):
        if not failed_indices:
            break
            
        # Try to embed failed items
        batch_to_retry = [(i, texts[i]) for i in failed_indices]
        newly_failed = []
        
        for idx, text in batch_to_retry:
            try:
                embeddings[idx] = self._generate_single_embedding(text)
            except Exception as e:
                newly_failed.append(idx)
                if attempt == self.max_retries:
                    logger.error(f"Failed to embed text at index {idx} after {attempt + 1} attempts")
        
        failed_indices = newly_failed
        
        if failed_indices and attempt < self.max_retries:
            delay = min(2 ** attempt, 8)
            time.sleep(delay)
    
    # Filter out None values (permanent failures)
    return [emb for emb in embeddings if emb is not None]
```
