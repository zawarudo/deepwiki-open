### 2. Create Shell Script `scripts/run_e2e_tests.sh`
```bash
#!/bin/bash
# End-to-end test runner with environment setup

echo "=== DeepWiki Embedding Pipeline E2E Test ==="
echo "Testing fixes for empty vector and dimension issues"
echo ""

# Check environment
if [ -z "$GOOGLE_API_KEY" ]; then
    echo "Warning: GOOGLE_API_KEY not set. Using mock mode."
    export USE_MOCK_EMBEDDINGS=true
fi

# Run tests with detailed output
echo "1. Testing empty embedding handling..."
pytest test/test_e2e_embedding_pipeline.py::TestE2EEmbeddingPipeline::test_complete_pipeline_with_mixed_inputs -v

echo ""
echo "2. Testing dimension validation..."
pytest test/test_e2e_embedding_pipeline.py::TestE2EEmbeddingPipeline::test_dimension_consistency_maintained -v

echo ""
echo "3. Testing error recovery..."
pytest test/test_e2e_embedding_pipeline.py::TestE2EEmbeddingPipeline::test_pipeline_recovery_from_api_failures -v

echo ""
echo "4. Testing error reporting..."
pytest test/test_e2e_embedding_pipeline.py::TestE2EEmbeddingPipeline::test_error_reporting_in_pipeline -v

# Summary
echo ""
echo "=== Test Summary ==="
pytest test/test_e2e_embedding_pipeline.py --tb=no -q

# Check for any remaining issues
echo ""
echo "Checking for empty vectors in output..."
grep -r "\\[\\]" test/test_output/ 2>/dev/null || echo "✅ No empty vectors found"

echo ""
echo "E2E test complete!"
```
