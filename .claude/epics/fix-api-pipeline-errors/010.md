#!/bin/bash
# Complete test suite runner

echo "=== DeepWiki Embedding Fix - Full Test Suite ==="
date
echo ""

# Test results file
RESULTS_FILE="test-results/full-test-results-$(date +%Y%m%d-%H%M%S).md"
mkdir -p test-results

# Function to run and report test
run_test() {
    local test_name=$1
    local test_command=$2
    echo "Running: $test_name"
    echo "## $test_name" >> $RESULTS_FILE
    echo '```' >> $RESULTS_FILE
    if $test_command >> $RESULTS_FILE 2>&1; then
        echo "✅ PASSED"
        echo "Result: PASSED" >> $RESULTS_FILE
    else
        echo "❌ FAILED"
        echo "Result: FAILED" >> $RESULTS_FILE
    fi
    echo '```' >> $RESULTS_FILE
    echo "" >> $RESULTS_FILE
}

# Run all test suites
echo "# Full Test Results - $(date)" > $RESULTS_FILE
echo "" >> $RESULTS_FILE

run_test "Task 001: Empty Embedding Validation" "pytest test/test_empty_embedding_validation.py -v"
run_test "Task 003: Dimension Consistency" "pytest test/test_dimension_consistency.py -v"
run_test "Task 005: Batch Retry Logic" "pytest test/test_batch_retry_logic.py -v"
run_test "Task 007: Error Reporting" "pytest test/test_error_reporting.py -v"
run_test "Task 009: End-to-End Pipeline" "pytest test/test_e2e_embedding_pipeline.py -v"

# Performance tests
run_test "Performance: Embedding Speed" "python test/test_performance.py"
run_test "Performance: Memory Usage" "python test/test_memory.py"

# Integration tests
run_test "Integration: FAISS Index Creation" "python test/test_faiss_integration.py"
run_test "Integration: WebSocket Pipeline" "python test/test_websocket_integration.py"

echo ""
echo "=== Test Summary ==="
cat $RESULTS_FILE | grep "Result:" | sort | uniq -c

echo ""
echo "Full results saved to: $RESULTS_FILE"
```
